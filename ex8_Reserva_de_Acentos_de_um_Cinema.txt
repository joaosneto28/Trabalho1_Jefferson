package com.mycompany.mavenproject1;
import java.util.Scanner;
/**
 *
 * @author fef
 */
public class ex8_Reserva_de_Acentos_de_um_Cinema {

    /**
     * @param args the command line arguments
     */
    public static void main(String[] args) {
        Scanner ler = new Scanner(System.in);
        // Matriz 10x12: 0 = Livre, 1 = Ocupado
        int[][] cinema = new int[10][12];
        int opcao = 0;
        while (opcao != 6) {
            System.out.println("\n--- CINEMA - MENU DE OPCOES ---");
            System.out.println("1. Reservar um assento");
            System.out.println("2. Cancelar uma reserva");
            System.out.println("3. Exibir o mapa dos assentos");
            System.out.println("4. Mostrar quantidade de assentos livres e ocupados");
            System.out.println("5. Encontrar sequencia de assentos livres");
            System.out.println("6. Encerrar");
            System.out.print("Escolha uma opcao: ");
            opcao = ler.nextInt();
            switch (opcao) {
                case 1:
                    System.out.print("Fileira (0-9): ");
                    int fRes = ler.nextInt();
                    System.out.print("Assento (0-11): ");
                    int aRes = ler.nextInt();
                    if (fRes < 0 || fRes >= 10 || aRes < 0 || aRes >= 12) {
                        System.out.println("Posicao invalida!");
                    } else if (cinema[fRes][aRes] == 1) {
                        System.out.println("Assento já esta ocupado!");
                    } else {
                        cinema[fRes][aRes] = 1;
                        System.out.println("Assento reservado com sucesso!");
                    }
                    break;
                case 2:
                    System.out.print("Fileira (0-9): ");
                    int fCanc = ler.nextInt();
                    System.out.print("Assento (0-11): ");
                    int aCanc = ler.nextInt();
                    if (fCanc < 0 || fCanc >= 10 || aCanc < 0 || aCanc >= 12) {
                        System.out.println("Posicao invalida!");
                    } else if (cinema[fCanc][aCanc] == 0) {
                        System.out.println("Assento ja está livre!");
                    } else {
                        cinema[fCanc][aCanc] = 0;
                        System.out.println("Reserva cancelada com sucesso!");
                    }
                    break;
                case 3:
                    System.out.println("\nMAPA DE ASSENTOS (0: Livre | 1: Ocupado)");
                    System.out.print("      ");
                    for (int j = 0; j < 12; j++) System.out.printf("[%02d]", j);
                    System.out.println();
                    for (int i = 0; i < 10; i++) {
                        System.out.printf("F%02d:  ", i);
                        for (int j = 0; j < 12; j++) {
                            System.out.print(" " + cinema[i][j] + "  ");
                        }
                        System.out.println();
                    }
                    break;
                case 4:
                    int livres = 0;
                    int ocupados = 0;
                    for (int i = 0; i < 10; i++) {
                        for (int j = 0; j < 12; j++) {
                            if (cinema[i][j] == 0) livres++;
                            else ocupados++;
                        }
                    }
                    System.out.println("Assentos livres: " + livres);
                    System.out.println("Assentos ocupados: " + ocupados);
                    break;
                case 5:
                    System.out.print("Quantidade de pessoas (assentos consecutivos): ");
                    int qtd = ler.nextInt();
                    if (qtd <= 0 || qtd > 12) {
                        System.out.println("Quantidade invalida.");
                        break;
                    }
                    boolean encontrou = false;
                    for (int i = 0; i < 10; i++) {
                        int consecutivos = 0;
                        int inicioAssento = -1;
                        for (int j = 0; j < 12; j++) {
                            if (cinema[i][j] == 0) {
                                if (consecutivos == 0) inicioAssento = j;
                                consecutivos++;
                                if (consecutivos == qtd) {
                                    System.out.printf("Encontrada sequencia na Fileira %d, do assento %d ao %d.\n",
                                            i, inicioAssento, j);
                                    encontrou = true;
                                    break; // Sai do laço da fileira
                                }
                            } else {
                                consecutivos = 0;
                            }
                        }
                    }
                    if (!encontrou) {
                        System.out.println("Nao foi encontrada nenhuma sequencia com " + qtd + " assentos livres na mesma fileira.");
                    }
                    break;
                case 6:
                    System.out.println("Encerrando o sistema...");
                    break;
                default:
                    System.out.println("Opção invalida!");
            }
        }
    }  
}
